
#[cfg(test)]
mod sift_fake {
    use super::{CaptureError, ScreenLife, Session};
    use std::sync::atomic::{AtomicBool, Ordering};
    use std::sync::Arc;
    use std::time::Duration;

    struct Fake {
        end_called: Arc<AtomicBool>,
        dropped: Arc<AtomicBool>,
    }

    impl Session for Fake {
        fn finished(&self) -> bool {
            false
        }

        fn end(self) -> Result<(), CaptureError> {
            self.end_called.store(true, Ordering::SeqCst);
            Ok(())
        }
    }

    impl Drop for Fake {
        fn drop(&mut self) {
            self.dropped.store(true, Ordering::SeqCst);
        }
    }

    #[test]
    fn sift_fake_session_when_the_stop_thread_cannot_be_spawned() {
        let end_called = Arc::new(AtomicBool::new(false));
        let dropped = Arc::new(AtomicBool::new(false));
        let mut life = ScreenLife::default();
        life.started(Fake {
            end_called: Arc::clone(&end_called),
            dropped: Arc::clone(&dropped),
        });
        let busy_before = life.is_busy();
        let first = life.shutdown(Duration::from_millis(300));
        std::thread::sleep(Duration::from_millis(100));
        println!(
            "VI7 spawnfail-fake fail_env_set={} busy_before={busy_before} first={first:?} busy={} state={:?} end_called={} session_dropped={}",
            std::env::var_os("HALVO_SIFT_FAIL_STOP_SPAWN").is_some(),
            life.is_busy(),
            life.state(false, false),
            end_called.load(Ordering::SeqCst),
            dropped.load(Ordering::SeqCst)
        );
        let second = life.shutdown(Duration::from_millis(300));
        println!(
            "VI7 spawnfail-fake second={second:?} busy={} state={:?}",
            life.is_busy(),
            life.state(false, false)
        );
    }
}
